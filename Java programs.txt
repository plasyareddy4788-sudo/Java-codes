import java.util.*;
public class Main {
    public static void main(String[] args) {
        String a = "AaBbCc";
        char[] s = a.toCharArray();
        int i=0;
        for ( i=0; i < s.length; i++)
        {
            if(s[i]>=65 && s[i]<92)
            {
                s[i]=(char)(s[i]+32);
            }
        }
        for ( i=0; i < s.length; i++)
        {
            System.out.print(s[i]);
        }
    }
}            aabbcc
